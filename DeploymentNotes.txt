There are four parts to this software:

recovery_worker.py (a service that heals failures)
worker.py (performs the main research)
FAST API (exposes an API to control the researcher)
Frontend (a web UI for admins and users)

This app uses a MSSQL backend over TCP/IP and requires an SQL account.
Create a DB, call it 'Researcher' and allow the user account access.
Download:
ODBC Driver 18 for SQL Server
https://learn.microsoft.com/en-us/sql/connect/odbc/download-odbc-driver-for-sql-server?view=sql-server-ver17
Add the SQL account details to .env and run create_database.py.

8. Create a file called LGResearchRecovery.xml in the LGResearchRecovery folder.  Replace
the text with that provided below:

<service>
  <id>LGResearchRecovery</id>
  <name>LG Research Recovery Worker</name>

  <description>
    Continuous recovery coordinator for LG Researcher.
  </description>

  <executable>%BASE%\app\.venv\Scripts\python.exe</executable>

  <arguments>-u "%BASE%\app\recovery_worker.py" --continuous --interval 10 --limit 25 --health-file "%BASE%\health\recovery-health.json"</arguments>

  <workingdirectory>%BASE%\app</workingdirectory>

  <env name="PYTHONUNBUFFERED" value="1" />
  <env name="PYTHONUTF8" value="1" />

  <startmode>Automatic</startmode>
  <delayedAutoStart>true</delayedAutoStart>

  <stoptimeout>5 min</stoptimeout>

  <onfailure action="restart" delay="10 sec"/>
  <resetfailure>1 hour</resetfailure>

  <logpath>%BASE%\logs</logpath>
  <log mode="roll-by-size">
    <sizeThreshold>10485760</sizeThreshold>
    <keepFiles>10</keepFiles>
  </log>
</service>

Useful SQL

Purge everything related to a job


SET NOCOUNT ON;
SET XACT_ABORT ON;

DECLARE @JobId VARCHAR(36) =
    'b9e82da9-8777-4504-acd7-69dd80640a89';

DECLARE @SessionId VARCHAR(36);

BEGIN TRY

    BEGIN TRANSACTION;

    --------------------------------------------------
    -- Find the job and its research session.
    --------------------------------------------------

    SELECT
        @SessionId = research_session_id
    FROM dbo.research_jobs WITH (
        UPDLOCK,
        HOLDLOCK
    )
    WHERE id = @JobId;

    IF @@ROWCOUNT = 0
    BEGIN
        THROW 50001,
            'Research job was not found.',
            1;
    END;

    --------------------------------------------------
    -- Reject active execution.
    --------------------------------------------------

    IF EXISTS (
        SELECT 1
        FROM dbo.research_jobs
        WHERE id = @JobId
          AND execution_owner IS NOT NULL
          AND lease_expires_at > SYSUTCDATETIME()
    )
    BEGIN
        THROW 50002,
            'Research job has an active execution lease.',
            1;
    END;

    IF @SessionId IS NOT NULL
       AND EXISTS (
            SELECT 1
            FROM dbo.research_sessions
            WHERE id = @SessionId
              AND execution_owner IS NOT NULL
              AND execution_lease_expires_at > SYSUTCDATETIME()
       )
    BEGIN
        THROW 50003,
            'Research session has an active execution lease.',
            1;
    END;

    --------------------------------------------------
    -- Protect research sessions belonging to
    -- another job.
    --------------------------------------------------

    IF @SessionId IS NOT NULL
       AND EXISTS (
            SELECT 1
            FROM dbo.research_jobs
            WHERE research_session_id = @SessionId
              AND id <> @JobId
       )
    BEGIN
        THROW 50004,
            'Research session is referenced by another job.',
            1;
    END;

    PRINT 'Purging job: ' + @JobId;

    --------------------------------------------------
    -- Delete all research data if a session exists.
    --------------------------------------------------

    IF @SessionId IS NOT NULL
    BEGIN

        PRINT 'Research session: ' + @SessionId;

        --------------------------------------------------
        -- Report provenance.
        --------------------------------------------------

        DELETE rsc
        FROM dbo.report_section_claims AS rsc
        INNER JOIN dbo.report_sections AS rs
            ON rs.id = rsc.report_section_id
        INNER JOIN dbo.research_reports AS rr
            ON rr.id = rs.research_report_id
        WHERE rr.research_session_id = @SessionId;

        --------------------------------------------------
        -- Report limitations.
        --------------------------------------------------

        DELETE rl
        FROM dbo.report_limitations AS rl
        INNER JOIN dbo.research_reports AS rr
            ON rr.id = rl.research_report_id
        WHERE rr.research_session_id = @SessionId;

        --------------------------------------------------
        -- Report sections.
        --------------------------------------------------

        DELETE rs
        FROM dbo.report_sections AS rs
        INNER JOIN dbo.research_reports AS rr
            ON rr.id = rs.research_report_id
        WHERE rr.research_session_id = @SessionId;

        --------------------------------------------------
        -- Reports.
        --------------------------------------------------

        DELETE FROM dbo.research_reports
        WHERE research_session_id = @SessionId;

        --------------------------------------------------
        -- Reviewed claim support.
        --------------------------------------------------

        DELETE rcs
        FROM dbo.reviewed_claim_support AS rcs
        INNER JOIN dbo.reviewed_claims AS rc
            ON rc.id = rcs.reviewed_claim_id
        INNER JOIN dbo.question_reviews AS qr
            ON qr.id = rc.question_review_id
        INNER JOIN dbo.research_questions AS rq
            ON rq.id = qr.research_question_id
        WHERE rq.research_session_id = @SessionId;

        --------------------------------------------------
        -- Reviewed claim contradictions.
        --------------------------------------------------

        DELETE rcc
        FROM dbo.reviewed_claim_contradictions AS rcc
        INNER JOIN dbo.reviewed_claims AS rc
            ON rc.id = rcc.reviewed_claim_id
        INNER JOIN dbo.question_reviews AS qr
            ON qr.id = rc.question_review_id
        INNER JOIN dbo.research_questions AS rq
            ON rq.id = qr.research_question_id
        WHERE rq.research_session_id = @SessionId;

        --------------------------------------------------
        -- Reviewed claims.
        --------------------------------------------------

        DELETE rc
        FROM dbo.reviewed_claims AS rc
        INNER JOIN dbo.question_reviews AS qr
            ON qr.id = rc.question_review_id
        INNER JOIN dbo.research_questions AS rq
            ON rq.id = qr.research_question_id
        WHERE rq.research_session_id = @SessionId;

        --------------------------------------------------
        -- Review contradictions.
        --------------------------------------------------

        DELETE c
        FROM dbo.review_contradictions AS c
        INNER JOIN dbo.question_reviews AS qr
            ON qr.id = c.question_review_id
        INNER JOIN dbo.research_questions AS rq
            ON rq.id = qr.research_question_id
        WHERE rq.research_session_id = @SessionId;

        --------------------------------------------------
        -- Review evidence gaps.
        --------------------------------------------------

        DELETE g
        FROM dbo.review_evidence_gaps AS g
        INNER JOIN dbo.question_reviews AS qr
            ON qr.id = g.question_review_id
        INNER JOIN dbo.research_questions AS rq
            ON rq.id = qr.research_question_id
        WHERE rq.research_session_id = @SessionId;

        --------------------------------------------------
        -- Question reviews.
        --------------------------------------------------

        DELETE qr
        FROM dbo.question_reviews AS qr
        INNER JOIN dbo.research_questions AS rq
            ON rq.id = qr.research_question_id
        WHERE rq.research_session_id = @SessionId;

        --------------------------------------------------
        -- Findings.
        --------------------------------------------------

        DELETE f
        FROM dbo.findings AS f
        INNER JOIN dbo.research_questions AS rq
            ON rq.id = f.research_question_id
        WHERE rq.research_session_id = @SessionId;

        --------------------------------------------------
        -- Source summaries.
        --------------------------------------------------

        DELETE ss
        FROM dbo.source_summaries AS ss
        INNER JOIN dbo.research_questions AS rq
            ON rq.id = ss.research_question_id
        WHERE rq.research_session_id = @SessionId;

        --------------------------------------------------
        -- Sources.
        --------------------------------------------------

        DELETE FROM dbo.sources
        WHERE research_session_id = @SessionId;

        --------------------------------------------------
        -- Search results.
        --------------------------------------------------

        DELETE FROM dbo.web_search_results
        WHERE research_session_id = @SessionId;

        --------------------------------------------------
        -- Search attempts.
        --------------------------------------------------

        DELETE FROM dbo.search_attempts
        WHERE research_session_id = @SessionId;

        --------------------------------------------------
        -- Research questions.
        --------------------------------------------------

        DELETE FROM dbo.research_questions
        WHERE research_session_id = @SessionId;

    END
    ELSE
    BEGIN

        PRINT 'Job has no research session.';
        PRINT 'Deleting job-level records only.';

    END;

    --------------------------------------------------
    -- Delete job questions.
    --------------------------------------------------

    DELETE FROM dbo.research_job_questions
    WHERE research_job_id = @JobId;

    --------------------------------------------------
    -- Delete job.
    --------------------------------------------------

    DELETE FROM dbo.research_jobs
    WHERE id = @JobId;

    --------------------------------------------------
    -- Delete research session if present.
    --------------------------------------------------

    IF @SessionId IS NOT NULL
    BEGIN

        DELETE FROM dbo.research_sessions
        WHERE id = @SessionId;

    END;

    COMMIT TRANSACTION;

    PRINT '';
    PRINT 'Research job purge completed successfully.';

END TRY

BEGIN CATCH

    IF @@TRANCOUNT > 0
    BEGIN
        ROLLBACK TRANSACTION;
    END;

    PRINT '';
    PRINT 'Research job purge FAILED.';
    PRINT ERROR_MESSAGE();

    THROW;

END CATCH;
